//@version=5
indicator("Correlerror (Fixed)", overlay = false)

// Inputs
LB = input.int(title = "Long Bar", minval = 3, maxval = 600, defval = 20)
SB = input.int(title = "Short Bar", minval = 3, maxval = 600, defval = 50)

// Basic Series
h = high
l = low
o = open
c = close
v = volume

// Range calculations
HL = ta.highest(h, SB) - ta.lowest(l, SB)
MV = ta.highest(v, SB) - ta.lowest(v, SB)

// Avoid divide-by-zero
HL := HL == 0 ? 1 : HL
MV := MV == 0 ? 1 : MV

// Ratio Logic
ratio = ((h - l) / HL) * (v / MV) * (c / o)

// Linear Regression Components
aa = ta.linreg(ratio, SB, 0)
bb = (ta.linreg(ratio, SB, 0) - ta.linreg(ratio, SB, 0)[SB]) / SB

res = ratio - (aa + bb * ratio)

// Correlation
corx = ta.correlation(ratio, ratio - res, LB)

// Z-Score
Z = (corx - ta.sma(corx, SB)) / ta.stdev(corx, SB)

// Correlation Index
CI = Z * 10

// Background Signals
bgcolor(ta.cross(CI, -19) ? color.new(color.green, 0) : na)
bgcolor(ta.cross(CI, 19) ? color.new(color.red, 0) : na)

// Plot
plot(CI, color = color.white, linewidth = 2)
